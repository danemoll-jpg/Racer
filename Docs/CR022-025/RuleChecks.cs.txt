var report = new System.Collections.Generic.List<string>();
void Check(bool value,string name){report.Add((value?"PASS ":"FAIL ")+name);}

p.Cross(1,true,13);Check(p.NextGate==2,"repeated gate no progress");p.Cross(2,false,14);Check(p.NextGate==2&&p.LapValid,"wrong-way no progress, recovery allowed");

p.Miss(1,5);p.Miss(2,20);p.Cross(3,true,30);p.Cross(0,true,40);Check(p.Finished&&p.MissedGates==3&&p.PenaltySeconds==30,"multiple independent misses");Check(p.RaceTime(100)==30&&p.AdjustedTime(100)==60,"finish locks driving time; adjustment separate");
p.Restart();Check(p.PenaltySeconds==0&&p.MissedGates==0&&p.Penalties.Count==0&&!p.Started,"restart clears penalties and progress");
p.Cross(0,true,0);p.Miss(1,5);p.ResetToGrid();Check(p.NextGate==0&&!p.LapActive&&p.PenaltySeconds==5,"manual reset abandons lap, retains race penalty");p.Cross(0,true,20);Check(p.NextGate==1&&p.LapActive,"reset permits fresh lap");
